## **BACKEND PROMPT: WebSocket Token Auth + ACK Reliability**

### **Context**
- **Problem**: WS auth uses only `?role=reviewer&user_id=2` (spoofable). Broadcasts are fire-and-forget → messages lost on disconnect/server hiccup.
- **Fix (Level 2)**:
  1. **JWT token auth** on WS connect
  2. **Acknowledged broadcast**: `msg_id` (UUID), client ACKs, server retries unacked (max 3x, 2s backoff)
  3. **Client dedup**: `msg_id` tracking
  4. **Reconnection**: REST fallback for missed messages

#### 1. `app/core/security.py` (NEW FILE)
```python
import jwt
from datetime import datetime, timedelta
from app.core.config import settings

SECRET_KEY = settings.SECRET_KEY  # add SECRET_KEY to .env
ALGORITHM = "HS256"
TOKEN_TTL_MINUTES = 5

def create_ws_token(user_id: int, role: str) -> str:
    """Short-lived JWT for WS auth (5 min)."""
    payload = {
        "user_id": user_id,
        "role": role,
        "exp": datetime.utcnow() + timedelta(minutes=TOKEN_TTL_MINUTES),
        "type": "ws"
    }
    return jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

def decode_ws_token(token: str) -> dict:
    """Validate JWT, return {user_id, role}. Raises on invalid/expired."""
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        if payload.get("type") != "ws":
            raise ValueError("Invalid token type")
        return {"user_id": payload["user_id"], "role": payload["role"]}
    except jwt.ExpiredSignatureError:
        raise ValueError("Token expired")
    except jwt.InvalidTokenError:
        raise ValueError("Invalid token")
```

```python
from fastapi import APIRouter, HTTPException
from app.core.security import create_ws_token

router = APIRouter(prefix="/api/v1", tags=["auth"])

@router.post("/auth/ws-token")
async def get_ws_token(request: dict):
    """Returns a short-lived JWT for WS auth.
    Request: {"user_id": 2, "role": "reviewer"}
    Response: {"token": "...", "expires_in": 300}
    """
    user_id = request.get("user_id")
    role = request.get("role")
    if not user_id or role not in ("worker", "reviewer", "admin"):
        raise HTTPException(status_code=400, detail="user_id and valid role required")
    token = create_ws_token(int(user_id), role)
    return {"token": token, "expires_in": 300}
```

#### 4. `app/core/websocket_manager.py` — Major upgrade
```python
import json
import uuid
import asyncio
import logging
from datetime import datetime
from typing import Dict, Set, Optional, Any, List
from fastapi import WebSocket

logger = logging.getLogger(__name__)

class PendingMessage:
    """Track unacked message for retry."""
    def __init__(self, msg_id: str, data: str, max_retries: int = 3, retry_delay: float = 2.0):
        self.msg_id = msg_id
        self.data = data
        self.retries = 0
        self.max_retries = max_retries
        self.retry_delay = retry_delay

    def should_retry(self) -> bool:
        return self.retries < self.max_retries

    def increment(self):
        self.retries += 1


class WebSocketManager:
    def __init__(self):
        self.connections: Dict[str, Set[WebSocket]] = {
            "worker": set(), "reviewer": set(), "admin": set(),
        }
        self.user_connections: Dict[int, WebSocket] = {}
        self.pending: Dict[WebSocket, Dict[str, PendingMessage]] = {}  # ws -> msg_id -> msg
        self.ack_lock = asyncio.Lock()

    async def connect(self, websocket: WebSocket, role: str, user_id: int):
        await websocket.accept()
        self.connections.setdefault(role, set()).add(websocket)
        self.user_connections[user_id] = websocket
        self.pending[websocket] = {}
        logger.info(f"WS connected: user={user_id}, role={role}")

    def disconnect(self, websocket: WebSocket, role: str, user_id: int):
        self.connections.get(role, set()).discard(websocket)
        self.user_connections.pop(user_id, None)
        self.pending.pop(websocket, None)
        logger.info(f"WS disconnected: user={user_id}, role={role}")

    def _add_msg_id(self, msg: dict) -> dict:
        """Attach msg_id to outgoing message if missing."""
        if "msg_id" not in msg:
            msg["msg_id"] = str(uuid.uuid4())
        return msg

    async def send_json(self, websocket: WebSocket, msg: dict):
        """Send with ACK tracking + retry loop."""
        msg = self._add_msg_id(msg)
        data = json.dumps(msg)
        pending = PendingMessage(msg["msg_id"], data)
        self.pending[websocket][msg["msg_id"]] = pending

        for attempt in range(pending.max_retries):
            try:
                await websocket.send_text(data)
                # Wait for ACK (timeout 3s)
                try:
                    ack_ok = await self._wait_for_ack(websocket, msg["msg_id"], timeout=3.0)
                    if ack_ok:
                        self.pending[websocket].pop(msg["msg_id"], None)
                        return
                except asyncio.TimeoutError:
                    pending.increment()
                    logger.warning(f"Msg {msg['msg_id']} not acked (attempt {attempt+1})")
                    continue
            except Exception as e:
                logger.error(f"Send failed for {msg['msg_id']}: {e}")
                break

        # Give up after retries
        logger.error(f"Msg {msg['msg_id']} dropped after {pending.max_retries} attempts")

    async def _wait_for_ack(self, websocket: WebSocket, msg_id: str, timeout: float = 3.0) -> bool:
        """Wait up to timeout for client ACK for this msg_id."""
        deadline = asyncio.get_event_loop().time() + timeout
        # NOTE: relies on the WS endpoint receiving ACK messages and calling handle_ack
        while asyncio.get_event_loop().time() < deadline:
            remaining = deadline - asyncio.get_event_loop().time()
            try:
                raw = await asyncio.wait_for(websocket.receive_text(), timeout=remaining)
                try:
                    ack = json.loads(raw)
                except json.JSONDecodeError:
                    continue
                if ack.get("type") == "ack" and ack.get("msg_id") == msg_id:
                    return True
                # ignore other messages while waiting
            except asyncio.TimeoutError:
                return False
            except Exception:
                return False
        return False

    # --- Broadcast with ACK ---
    async def broadcast_to_role(self, role: str, message: dict):
        msg = self._add_msg_id(message)
        tasks = []
        dead = set()
        for ws in self.connections.get(role, set()):
            try:
                tasks.append(asyncio.create_task(self.send_json(ws, msg)))
            except Exception:
                dead.add(ws)
        for ws in dead:
            self.connections[role].discard(ws)
        # Wait for all ACK send tasks (with timeout, don't block forever)
        if tasks:
            await asyncio.gather(*tasks, return_exceptions=True)

    async def broadcast_all(self, message: dict):
        for role in self.connections:
            await self.broadcast_to_role(role, message)

    async def send_to_user(self, user_id: int, message: dict):
        ws = self.user_connections.get(user_id)
        if ws:
            await self.send_json(ws, message)
```

#### 5. `app/api/v1/routes_escalation.py` — Upgrade WS endpoint
```python
@router.websocket("/ws/escalations")
async def websocket_escalations(
    websocket: WebSocket,
    token: str = Query(...),  # <-- CHANGED from role+user_id
    db: AsyncSession = Depends(get_db)
):
    # Auth via JWT
    try:
        auth = decode_ws_token(token)
    except ValueError as e:
        await websocket.close(code=4001, reason=str(e))
        return
    role, user_id = auth["role"], auth["user_id"]

    await ws_manager.connect(websocket, role, user_id)
    try:
        while True:
            data = await websocket.receive_text()
            try:
                msg = json.loads(data)
            except json.JSONDecodeError:
                continue

            # Handle ACK — remove from retry queue
            if msg.get("type") == "ack":
                await ws_manager._ack_received(websocket, msg.get("msg_id"))
                continue

            # Handle ping heartbeat
            if msg.get("type") == "ping":
                await websocket.send_text(json.dumps({"type": "pong", "msg_id": str(uuid.uuid4()), "timestamp": datetime.utcnow().isoformat()}))
                continue

    except WebSocketDisconnect:
        ws_manager.disconnect(websocket, role, user_id)
    except Exception as e:
        logger.error(f"WS error: {e}")
        ws_manager.disconnect(websocket, role, user_id)
```

### **Expected Behavior After Fix**

| Scenario | Result |
|----------|--------|
| Connect without token / invalid token | `WebSocket.close(4001)` |
| Expired token | `close(4001, "Token expired")` — client re-fetches + reconnects |
| Message sent but client offline | Server retries 3x (2s apart), then drops |
| Client ACKs | Removed from retry queue immediately |
| Client reconnects after miss | Missed messages NOT replayed (Level 2 limits this) — REST fallback: `GET /escalations/pending` refetch |
