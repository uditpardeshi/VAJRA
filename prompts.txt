### 3. Model Router (`app/core/model_router.py`) — **NEW: Multi-model client**
```python
import httpx
import json
import logging
from typing import Optional, List, Dict, Any
from app.core.config import settings

logger = logging.getLogger(__name__)

class ModelRouter:
    """
    Routes requests to appropriate model via ngrok tunnel.
    Supports: vision, text, code, embedding tasks.
    """
    
    def __init__(self):
        base = settings.NGROK_OLLAMA_URL.rstrip("/")
        if not base:
            raise RuntimeError("NGROK_OLLAMA_URL not set in .env")
        self.base_url = base
        self.registry = settings.model_registry
        self.timeout = 180.0  # Colab cold start
        
        self.headers = {"Content-Type": "application/json"}
        if settings.NGROK_SKIP_WARNING:
            self.headers["ngrok-skip-browser-warning"] = "true"

    async def _post(self, endpoint: str, payload: dict) -> dict:
        async with httpx.AsyncClient(timeout=self.timeout, headers=self.headers) as client:
            resp = await client.post(f"{self.base_url}{endpoint}", json=payload)
            if resp.status_code == 403 and "ngrok" in resp.text.lower():
                raise RuntimeError("ngrok tunnel blocked (warning page?). Check NGROK_SKIP_WARNING.")
            resp.raise_for_status()
            return resp.json()

    def _get_model(self, task: str) -> str:
        model = self.registry.get(task)
        if not model:
            raise ValueError(f"No model configured for task: {task}")
        return model

    # ---------- VISION: Image + prompt → structured JSON ----------
    async def vision_inspect(
        self,
        image_base64: str,
        system_prompt: str,
        model: Optional[str] = None
    ) -> dict:
        payload = {
            "model": model or self._get_model("vision"),
            "prompt": system_prompt,
            "images": [image_base64],
            "stream": False,
            "format": "json",
            "options": {"temperature": 0.1, "num_predict": 1024}
        }
        data = await self._post("/api/generate", payload)
        raw = data.get("response", "{}")
        try:
            return json.loads(raw)
        except json.JSONDecodeError:
            logger.error(f"Vision model returned non-JSON: {raw[:200]}")
            return {"error": "parse_failed", "raw": raw[:500]}

    # ---------- TEXT: Prompt → string ----------
    async def text_chat(
        self,
        prompt: str,
        system: Optional[str] = None,
        model: Optional[str] = None,
        temperature: float = 0.2,
        max_tokens: int = 2048
    ) -> str:
        payload = {
            "model": model or self._get_model("text"),
            "prompt": prompt,
            "system": system,
            "stream": False,
            "options": {"temperature": temperature, "num_predict": max_tokens}
        }
        data = await self._post("/api/generate", payload)
        return data.get("response", "")

    # ---------- CODE: Prompt → code + execution result (future) ----------
    async def code_execute(
        self,
        prompt: str,
        system: Optional[str] = None,
        model: Optional[str] = None
    ) -> dict:
        """Generate code, return code string. Execution is separate (sandbox)."""
        payload = {
            "model": model or self._get_model("code"),
            "prompt": prompt,
            "system": system or "You are a Python expert. Generate clean, safe code only.",
            "stream": False,
            "options": {"temperature": 0.1, "num_predict": 2048}
        }
        data = await self._post("/api/generate", payload)
        return {"code": data.get("response", "")}

    # ---------- EMBEDDINGS: Text → vector (future) ----------
    async def embed(self, texts: List[str], model: Optional[str] = None) -> List[List[float]]:
        payload = {
            "model": model or self._get_model("embedding"),
            "input": texts
        }
        data = await self._post("/api/embed", payload)
        return data.get("embeddings", [])

    # ---------- HEALTH: List available models ----------
    async def list_models(self) -> List[dict]:
        data = await self._post("/api/tags", {})
        return data.get("models", [])

model_router = ModelRouter()
```

### 6. Inspection Service (`app/services/inspection_service.py`) — *updated to use model_router*
```python
import json
from datetime import datetime
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from app.models.tables import Inspection, Machine, AuditLog, Ticket, TicketStatus, Escalation, EscalationStatus
from app.core.model_router import model_router
from app.schemas.inspection import InspectRequest, InspectResponse, DefectLocation

async def create_inspection(
    db: AsyncSession,
    request: InspectRequest,
    user_id: int = 1
) -> InspectResponse:
    machine = await db.execute(
        select(Machine).where(Machine.machine_id == request.machine_id)
    )
    machine = machine.scalar_one_or_none()
    if not machine:
        raise ValueError(f"Machine {request.machine_id} not found")

    specs = json.loads(machine.specs_json) if machine.specs_json else {}

    # Build prompt
    system_prompt = f"""You are an expert industrial maintenance engineer analyzing equipment photos.
Machine: {machine.machine_id}
Key Specs: {json.dumps(specs, indent=2)}

Analyze the image for: wear, misalignment, leaks, cracks, abnormal vibration signs, safety hazards.
Return ONLY valid JSON with these exact keys:
- "finding": string (concise technical description)
- "confidence": float 0.0-1.0
- "defect_location": object with x,y,w,h normalized 0-1 (or null)
- "repair_steps": array of strings (ordered, actionable)
- "needs_escalation": boolean (true if safety-critical or confidence < 0.7)
"""
    if request.prompt_override:
        system_prompt += f"\nAdditional focus: {request.prompt_override}"

    # Call vision model via router
    ollama_result = await model_router.vision_inspect(
        image_base64=request.image_base64,
        system_prompt=system_prompt
    )

    # Handle parse errors
    if "error" in ollama_result:
        ollama_result = {
            "finding": "Model response parse failed",
            "confidence": 0.0,
            "defect_location": None,
            "repair_steps": [],
            "needs_escalation": True
        }

    # Normalize
    ollama_result.setdefault("finding", "No finding returned")
    ollama_result.setdefault("confidence", 0.5)
    ollama_result.setdefault("defect_location", None)
    ollama_result.setdefault("repair_steps", [])
    ollama_result.setdefault("needs_escalation", ollama_result["confidence"] < 0.7)

    # Persist
    defect_loc = ollama_result.get("defect_location")
    inspection = Inspection(
        machine_id=machine.id,
        finding=ollama_result["finding"],
        confidence=ollama_result["confidence"],
        defect_location=json.dumps(defect_loc) if defect_loc else None,
        repair_steps=json.dumps(ollama_result["repair_steps"]),
        needs_escalation=1 if ollama_result["needs_escalation"] else 0,
    )
    db.add(inspection)
    await db.flush()

    # Auto-ticket
    ticket = None
    if ollama_result["confidence"] > 0.5 and ollama_result["finding"].strip().lower() not in ("normal", "no issue", "ok"):
        ticket = Ticket(
            inspection_id=inspection.id,
            machine_id=machine.id,
            title=f"Inspection finding: {machine.machine_id}",
            description=ollama_result["finding"],
            status=TicketStatus.OPEN,
            priority=1 if ollama_result["needs_escalation"] else 2,
        )
        db.add(ticket)
        await db.flush()

        if ollama_result["needs_escalation"]:
            escalation = Escalation(
                ticket_id=ticket.id,
                reason="low_confidence" if ollama_result["confidence"] < 0.7 else "safety_critical",
                status=EscalationStatus.PENDING,
            )
            db.add(escalation)

    # Audit
    audit = AuditLog(
        user_id=user_id,
        action="inspect",
        resource_type="inspection",
        resource_id=inspection.id,
        details=json.dumps({
            "machine_id": request.machine_id,
            "confidence": ollama_result["confidence"],
            "needs_escalation": ollama_result["needs_escalation"],
            "ticket_created": ticket is not None
        })
    )
    db.add(audit)

    await db.commit()
    await db.refresh(inspection)

    return InspectResponse(
        inspection_id=inspection.id,
        machine_id=machine.machine_id,
        finding=inspection.finding,
        confidence=inspection.confidence,
        defect_location=DefectLocation(**json.loads(inspection.defect_location)) if inspection.defect_location else None,
        repair_steps=json.loads(inspection.repair_steps) if inspection.repair_steps else [],
        needs_escalation=bool(inspection.needs_escalation),
        created_at=inspection.created_at
    )

async def get_machines(db: AsyncSession) -> list[Machine]:
    from sqlalchemy import select
    result = await db.execute(select(Machine).order_by(Machine.machine_id))
    return result.scalars().all()

async def get_machine(db: AsyncSession, machine_id: str) -> Machine | None:
    from sqlalchemy import select
    result = await db.execute(select(Machine).where(Machine.machine_id == machine_id))
    return result.scalar_one_or_none()
```

### 7. API Routes (`app/api/v1/routes_inspect.py`) — *unchanged*
```python
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession
from app.core.database import get_db
from app.services.inspection_service import create_inspection, get_machines, get_machine
from app.schemas.inspection import InspectRequest, InspectResponse, MachineListResponse, MachineResponse

router = APIRouter(prefix="/api/v1", tags=["inspection"])

@router.post("/inspect", response_model=InspectResponse, status_code=status.HTTP_201_CREATED)
async def inspect_equipment(
    request: InspectRequest,
    db: AsyncSession = Depends(get_db)
):
    try:
        return await create_inspection(db, request)
    except ValueError as e:
        raise HTTPException(status_code=404, detail=str(e))
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Inspection failed: {str(e)}")

@router.get("/machines", response_model=MachineListResponse)
async def list_machines(db: AsyncSession = Depends(get_db)):
    machines = await get_machines(db)
    return MachineListResponse(machines=[MachineResponse.model_validate(m) for m in machines])

@router.get("/machines/{machine_id}", response_model=MachineResponse)
async def get_machine_detail(machine_id: str, db: AsyncSession = Depends(get_db)):
    machine = await get_machine(db, machine_id)
    if not machine:
        raise HTTPException(status_code=404, detail="Machine not found")
    return MachineResponse.model_validate(machine)
```
