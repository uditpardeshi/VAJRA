// ============================================================
// STATE
// ============================================================
let videoStream = null;
let ws = null;
let machines = [];

// ============================================================
// UTILITIES
// ============================================================
const $ = (id) => document.getElementById(id);
const base = () => $('baseUrl').value.replace(/\/+$/, '');
const wsBase = () => $('wsUrl').value.replace(/\/+$/, '');

function logWs(type, payload) {
  const entry = document.createElement('div');
  entry.className = `log-entry ${type}`;
  entry.innerHTML = `<strong>${new Date().toLocaleTimeString()}</strong> [${type.toUpperCase()}] ${JSON.stringify(payload).slice(0, 300)}`;
  $('wsLog').prepend(entry);
}

function setWsConnected(connected) {
  $('wsDot').classList.toggle('connected', connected);
  $('wsStatus').textContent = `WebSocket: ${connected ? 'Connected' : 'Disconnected'}`;
}

// ============================================================
// MACHINES LOADING
// ============================================================
async function loadMachines() {
  try {
    const res = await fetch(`${base()}/machines`);
    const data = await res.json();
    machines = data.machines || [];
    ['inspectMachine', 'chatMachine', 'agentMachine'].forEach(id => {
      const sel = $(id);
      const firstVal = sel.value;
      sel.innerHTML = '<option value="">Select Machine</option>' + 
        machines.map(m => `<option value="${m.machine_id}">${m.machine_id} — ${m.name}</option>`).join('');
      if (firstVal) sel.value = firstVal;
    });
  } catch (e) {
    console.error('Failed to load machines:', e);
  }
}

// ============================================================
// CAMERA
// ============================================================
async function startCamera() {
  try {
    videoStream = await navigator.mediaDevices.getUserMedia({ 
      video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } } 
    });
    $('video').srcObject = videoStream;
    $('video').classList.remove('hidden');
    $('canvas').classList.add('hidden');
  } catch (e) {
    alert('Camera access denied: ' + e.message);
  }
}

function stopCamera() {
  if (videoStream) {
    videoStream.getTracks().forEach(t => t.stop());
    videoStream = null;
  }
  $('video').classList.add('hidden');
  $('canvas').classList.remove('hidden');
}

function captureAndInspect() {
  const video = $('video');
  const canvas = $('canvas');
  const ctx = canvas.getContext('2d');
  
  canvas.width = video.videoWidth;
  canvas.height = video.videoHeight;
  ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
  
  video.classList.add('hidden');
  canvas.classList.remove('hidden');
  
  const base64 = canvas.toDataURL('image/jpeg', 0.8).split(',')[1];
  doInspect(base64);
}

async function doInspect(base64) {
  const machineId = $('inspectMachine').value;
  const prompt = $('inspectPrompt').value;
  
  if (!machineId) { alert('Select a machine'); return; }
  
  $('inspectResult').className = 'status warn';
  $('inspectResult').textContent = 'Inspecting...';
  $('inspectJson').textContent = '...';
  
  try {
    const res = await fetch(`${base()}/inspect`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ machine_id: machineId, image_base64: base64, prompt_override: prompt })
    });
    
    const data = await res.json();
    $('inspectJson').textContent = JSON.stringify(data, null, 2);
    
    if (res.ok) {
      $('inspectResult').className = 'status ok';
      $('inspectResult').textContent = `✅ Inspection #${data.inspection_id} — Confidence: ${(data.confidence*100).toFixed(0)}% — ${data.needs_escalation ? '⚠️ ESCALATED' : 'OK'}`;
      logWs('inspection', data);
    } else {
      $('inspectResult').className = 'status err';
      $('inspectResult').textContent = `❌ Error: ${data.detail || 'Unknown'}`;
    }
  } catch (e) {
    $('inspectResult').className = 'status err';
    $('inspectResult').textContent = `❌ Network error: ${e.message}`;
  }
}

// ============================================================
// CHAT / RAG
// ============================================================
async function sendChat() {
  const question = $('chatQuestion').value.trim();
  const machineId = $('chatMachine').value || null;
  const topK = parseInt($('chatTopK').value) || 3;
  
  if (!question) { alert('Enter a question'); return; }
  
  $('chatResult').className = 'status warn';
  $('chatResult').textContent = 'Querying...';
  $('chatJson').textContent = '...';
  
  try {
    const res = await fetch(`${base()}/chat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ question, machine_id: machineId, top_k: topK })
    });
    
    const data = await res.json();
    $('chatJson').textContent = JSON.stringify(data, null, 2);
    
    if (res.ok) {
      $('chatResult').className = 'status ok';
      $('chatResult').textContent = `✅ Answer (confidence: ${(data.confidence*100).toFixed(0)}%, ${data.citations.length} citations)`;
      logWs('chat', { question, citations: data.citations.length });
    } else {
      $('chatResult').className = 'status err';
      $('chatResult').textContent = `❌ Error: ${data.detail || 'Unknown'}`;
    }
  } catch (e) {
    $('chatResult').className = 'status err';
    $('chatResult').textContent = `❌ Network error: ${e.message}`;
  }
}

// ============================================================
// WEBSOCKET
// ============================================================
function connectWS() {
  const role = $('wsRole').value;
  const userId = $('wsUserId').value;
  const url = `${wsBase()}?role=${role}&user_id=${userId}`;
  
  ws = new WebSocket(url);
  
  ws.onopen = () => {
    setWsConnected(true);
    logWs('system', { event: 'connected', role, userId });
  };
  
  ws.onmessage = (event) => {
    try {
      const msg = JSON.parse(event.data);
      if (msg.type === 'pong') return;
      logWs(msg.type, msg.payload);
    } catch (e) {
      logWs('raw', event.data);
    }
  };
  
  ws.onclose = () => {
    setWsConnected(false);
    logWs('system', { event: 'disconnected' });
  };
  
  ws.onerror = (e) => {
    logWs('error', { message: 'WebSocket error' });
  };
  
  // Heartbeat
  setInterval(() => {
    if (ws && ws.readyState === WebSocket.OPEN) {
      ws.send(JSON.stringify({ ping: true }));
    }
  }, 30000);
}

function disconnectWS() {
  if (ws) { ws.close(); ws = null; }
  setWsConnected(false);
}

function clearWsLog() {
  $('wsLog').innerHTML = '<div class="log-entry">Log cleared</div>';
}

// ============================================================
// AGENT DOCUMENT ANALYSIS
// ============================================================
async function uploadAndAnalyze() {
  const file = $('agentFile').files[0];
  const machineId = $('agentMachine').value;
  const analysisType = $('agentType').value;
  
  if (!file) { alert('Select a file'); return; }
  if (!machineId) { alert('Select a machine'); return; }
  
  $('agentResult').className = 'status warn';
  $('agentResult').textContent = 'Analyzing... (this may take 30-60s on cold start)';
  $('agentJson').textContent = '...';
  
  const formData = new FormData();
  formData.append('file', file);
  formData.append('machine_id', machineId);
  formData.append('analysis_type', analysisType);
  
  try {
    const res = await fetch(`${base()}/analyze-doc`, {
      method: 'POST',
      body: formData
    });
    
    const data = await res.json();
    $('agentJson').textContent = JSON.stringify(data, null, 2);
    
    if (res.ok && data.status === 'completed') {
      $('agentResult').className = 'status ok';
      $('agentResult').textContent = `✅ Run #${data.run_id} — ${data.result.recommendation} — Deviation: ${data.result.deviation_mm ?? 'N/A'} mm`;
      logWs('agent', { run_id: data.run_id, recommendation: data.result.recommendation });
    } else if (res.ok && data.status === 'failed') {
      $('agentResult').className = 'status err';
      $('agentResult').textContent = `❌ Failed: ${data.result.error}`;
    } else {
      $('agentResult').className = 'status err';
      $('agentResult').textContent = `❌ HTTP ${res.status}: ${data.detail || 'Unknown'}`;
    }
  } catch (e) {
    $('agentResult').className = 'status err';
    $('agentResult').textContent = `❌ Network error: ${e.message}`;
  }
}

async function loadAgentRuns() {
  try {
    const res = await fetch(`${base()}/agent/runs?limit=20`);
    const data = await res.json();
    $('runsJson').textContent = JSON.stringify(data, null, 2);
  } catch (e) {
    $('runsJson').textContent = 'Error: ' + e.message;
  }
}

// ============================================================
// INIT
// ============================================================
loadMachines();
// Auto-connect WS if reviewer/admin
setTimeout(() => { if ($('wsRole').value !== 'engineer') connectWS(); }, 500);

// Cleanup on unload
window.addEventListener('beforeunload', () => { stopCamera(); disconnectWS(); });
</script>
</body>
</html>
